package intercambios

import "time"

// Usuario representa una persona registrada en la plataforma.
type Usuario struct {
	ID     uint   `gorm:"primaryKey" json:"id"`
	Nombre string `gorm:"not null" json:"nombre"`
	Correo string `gorm:"unique;not null" json:"correo"`
	Rol    string `gorm:"not null" json:"rol"`
}

// Libro representa un libro usado publicado por un usuario.
type Libro struct {
	ID         uint   `gorm:"primaryKey" json:"id"`
	Titulo     string `gorm:"not null" json:"titulo"`
	Autor      string `gorm:"not null" json:"autor"`
	Genero     string `gorm:"not null" json:"genero"`
	Disponible bool   `gorm:"not null;default:true" json:"disponible"`
	UsuarioID  uint   `gorm:"not null" json:"usuario_id"`
}

// Intercambio representa una solicitud de intercambio de un libro.
type Intercambio struct {
	ID            uint      `gorm:"primaryKey" json:"id"`
	LibroID       uint      `gorm:"not null" json:"libro_id"`
	SolicitanteID uint      `gorm:"not null" json:"solicitante_id"`
	Estado        string    `gorm:"not null;default:pendiente" json:"estado"`
	Creado        time.Time `gorm:"autoCreateTime" json:"creado"`
}

// Estados permitidos para un intercambio.
const (
	EstadoPendiente = "pendiente"
	EstadoAceptado  = "aceptado"
	EstadoEntregado = "entregado"
	EstadoCancelado = "cancelado"
)

// EstadoValido verifica que el estado recibido sea permitido.
func EstadoValido(estado string) bool {
	switch estado {
	case EstadoPendiente,
		EstadoAceptado,
		EstadoEntregado,
		EstadoCancelado:
		return true
	default:
		return false
	}
}
